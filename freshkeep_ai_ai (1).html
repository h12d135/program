STEP 1. Roboflow Inference API

식재료 신선도 분류 모델 호출 (Python)

import requests

# Roboflow 'food-freshness-classification-zu3j5-smeq4' API 설정
API_KEY = "YOUR_ROBOFLOW_API_KEY"
MODEL_ID = "food-freshness-classification-zu3j5-smeq4/1"
IMAGE_PATH = "apple_sample.jpg"

def predict_freshness(image_path):
    url = f"https://detect.roboflow.com/{MODEL_ID}?api_key={API_KEY}"
    with open(image_path, "rb") as img_file:
        response = requests.post(url, files={"file": img_file})
    
    result = response.json()
    for pred in result.get("predictions", []):
        label = pred["class"]        # 예: Fresh_Apple, Rotten_Banana
        confidence = pred["confidence"]
        print(f"인식 결과: {label} (신뢰도: {confidence:.2%})")
        return label, confidence
    return None, 0.0
STEP 2. 공공데이터포털 (data.go.kr) API

식품안전나라 소비기한·보관법 조회 (Python)

import requests

# 공공데이터포털(식품의약품안전처 소비기한 참고값 / 농식품올바로)
SERVICE_KEY = "YOUR_DATA_GO_KR_DECODED_KEY"
ENDPOINT = "https://apis.data.go.kr/1471000/FoodExpirationService/getFoodExpList"

def get_storage_guide(food_name, freshness_state):
    params = {
        "serviceKey": SERVICE_KEY,
        "type": "json",
        "PRDLST_NM": food_name,  # AI가 인식한 식재료명 매핑
        "numOfRows": 5
    }
    res = requests.get(ENDPOINT, params=params).json()
    items = res.get("body", {}).get("items", [])
    
    # 신선도 상태에 따른 잔여 소비기한 동적 보정 알고리즘
    multiplier = 1.0 if "Fresh" in freshness_state else 0.25
    return {
        "storage_method": items[0].get("CSTDY_MTH", "냉장보관(0~4℃)"),
        "adjusted_days": int(items[0].get("USE_BY_DAYS", 14) * multiplier)
    }
